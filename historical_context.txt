   424 ## 注意事项
   425 
   426 1. **密钥安全**：`INTEGRATIONS_API_KEY` 仅可在 Edge Function 服务端读取，严禁暴露到前端或客户端代码。
   427 
   428 2. **计费**：
   429    - 折后单价：**¥0.25 / 次**（原价 ¥0.30 / 次）
   430    - 每次调用均计费，避免在循环或高频场景中不必要地重复调用。
   431 
   432 3. **错误处理**：务必处理以下情况：
   433    - HTTP 429（配额超限）：已在 Edge Function 中透传，前端需提示用户稍后重试。
   434    - HTTP 402（余额不足）：已在 Edge Function 中透传，前端需提示充值。
   435    - `error_code !== 0`：OCR 服务自身错误，参考 `error_msg` 排查。
   436 
   437 4. **图片格式**：仅支持 jpg、jpeg、png、bmp 格式，上传前需客户端校验。
   438 
   439 5. **Base64 编码**：对于大图片，使用分块编码（chunkSize=8192）避免 `RangeError: Maximum call stack size exceeded`，禁止使用 `String.fromCharCode(...largeArray)` 展开写法。
   440 
   441 6. **MiniProgram 注意**：微信小程序中 `Taro.getFileSystemManager().readFile` 的 `encoding: "base64"` 返回的已是纯 Base64 字符串，无需再做 btoa 处理。
   442 
   443 7. **响应有效性判断**：不要用 `words_result_num > 0` 来判断是否有识别结果，应改为检查 `words_result` 数组本身（如 `words_result && words_result.length > 0`）。原因是当仅有段落结果（`paragraphs_result` 有值）而无逐行文字结果时，`words_result_num` 可能为 0，但 `words_result` 数组实际上可能并不为空。

function normalizeWord(input: string): string {

async function withConcurrencyLimit<T, R>(
  items: T[],
  limit: number,
  fn: (item: T) => Promise<R>
): Promise<R[]> {
  const results: R[] = new Array(items.length);
  let index = 0;

  async function worker() {
    while (index < items.length) {
      const currentIndex = index++;
      results[currentIndex] = await fn(items[currentIndex]);
    }
  }

  const workers: Promise<void>[] = [];
  for (let i = 0; i < Math.min(limit, items.length); i++) {
    workers.push(worker());
  }
  await Promise.all(workers);
  return results;
}

async function translateWords(
  words: string[],
  apiKey: string
): Promise<Record<string, string>> {
  const q = words.join("\n");
  const upstream = await fetch(TRANSLATION_API, {
    method: "POST",
    headers: {
      "Content-Type": "application/json;charset=utf-8",
      "X-Gateway-Authorization": `Bearer ${apiKey}`,
    },
    body: JSON.stringify({ q, from: "en", to: "zh" }),
  });

  if (upstream.status === 429 || upstream.status === 402) {
    const errText = await upstream.text();
    throw new Error(errText);
  }

  if (!upstream.ok) {
    throw new Error(`Translation upstream error: ${upstream.status}`);
  }

  const json = (await upstream.json()) as {
    error_code?: number;
    error_msg?: string;
    result?: TranslateResult;
  };

  if (json.error_code) {
    throw new Error(`Translation API error ${json.error_code}: ${json.error_msg}`);
  }

  const map: Record<string, string> = {};
  const transResult = json.result?.trans_result ?? [];
  for (let i = 0; i < words.length && i < transResult.length; i++) {
    map[words[i]] = transResult[i].dst;
  }
  return map;
}

async function fetchPhonetic(word: string): Promise<string> {
  try {
    const res = await fetch(
      `https://api.dictionaryapi.dev/api/v2/entries/en/${encodeURIComponent(word)}`,
      { method: "GET" }
    );
    if (!res.ok) return "";
    const entries = (await res.json()) as DictionaryEntry[];
    const entry = entries.find((e) => e.word.toLowerCase() === word.toLowerCase());
    if (!entry) return "";
    if (entry.phonetic) return entry.phonetic;
    for (const p of entry.phonetics ?? []) {
      if (p.text) return p.text;
    }
    return "";
  } catch {
    return "";
  }
}

/**
 * 将 File 对象转换为纯 Base64 字符串（不含 data URL 前缀）。
 * 采用分块读取，避免大图片触发 RangeError。
 */
export async function fileToBase64(file: File): Promise<string> {
  const buffer = await file.arrayBuffer();
  const bytes = new Uint8Array(buffer);
  const chunkSize = 8192;
  let binary = "";
  for (let i = 0; i < bytes.length; i += chunkSize) {
    binary += String.fromCharCode(...bytes.subarray(i, i + chunkSize));
  }
  return btoa(binary);
}

/**
 * 验证图片文件类型与大小。
 * 仅接受 jpg/jpeg/png/bmp，单张不超过 5MB。
 */
export function validateImage(file: File): { valid: boolean; message?: string } {
  const allowedTypes = ["image/jpeg", "image/jpg", "image/png", "image/bmp"];
  if (!allowedTypes.includes(file.type)) {
    return { valid: false, message: `不支持的文件类型：${file.name}` };
  }
  const maxSize = 5 * 1024 * 1024;
  if (file.size > maxSize) {
    return { valid: false, message: `图片大小超过 5MB：${file.name}` };
  }
  return { valid: true };
}

  if (words.length === 0) return;

  const header = ["单词", "音标", "中译"];
  const rows = words.map((item) => [

  const bom = "\uFEFF";

  const link = document.createElement("a");

  link.click();
  document.body.removeChild(link);

}

function escapeCsvField(value: string): string {

 * 按空格、换行、常见标点分割，过滤非字母字符并统一小写。

export function extractEnglishWords(texts: string[]): string[] {
  const words: string[] = [];
  for (const text of texts) {
    const segments = text
      .replace(/[\r\n]+/g, " ")
      .replace(/[^a-zA-Z\s'-]/g, " ")
      .split(/\s+/);
